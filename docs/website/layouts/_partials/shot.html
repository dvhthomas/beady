{{- /* One screenshot, by its name in assets/screenshots (scripts/screenshots.sh writes them). partial "shot.html" (dict "name" "board" "alt" "…" "caption" "…" "light" "ready-light" "frame" "window" "eager" true "sizes" "…") The PNGs are 2x; this serves WebP at a few widths and lets the browser choose. `light` names a second screenshot to show instead when the reader's system is in light mode. `frame` is "window" (a title bar is drawn on, since the snapshots are of the window's contents) or "panel" (a sheet or popover, floating); left out, full-window snapshots get a window. `crop` keeps only that many points from the top: the snapshot window is tall, and a list of eight beads leaves most of it empty. */ -}} {{- $img := resources.Get (printf "screenshots/%s.png" .name) }} {{- if not $img }}{{ errorf "shot: no screenshot named %q in assets/screenshots (see scripts/screenshots.sh)" .name }}{{ end }} {{- $wide := ge $img.Width 2400 }} {{- $crop := 0 }} {{- with .crop }}{{ $crop = mul (int .) 2 }}{{ end }} {{- if and (gt $crop 0) (lt $crop $img.Height) }}{{ $img = $img.Crop (printf "%dx%d TopLeft" $img.Width $crop) }}{{ end }} {{- $frame := .frame | default (cond $wide "window" "panel") }} {{- $sizes := .sizes | default (cond (eq $frame "window") "(min-width: 1180px) 1100px, 94vw" (printf "(min-width: %dpx) %dpx, 94vw" (div $img.Width 2) (div $img.Width 2))) }} {{- $widths := slice 720 1100 1600 2200 }}
